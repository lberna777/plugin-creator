# Template CMake — sostituisci i segnaposto @…@ con i valori dello spec, poi rinomina in CMakeLists.txt
cmake_minimum_required(VERSION 3.22)

if(APPLE)
    set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "" FORCE)   # universal
    set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "" FORCE)
endif()

project(@PLUGIN_NAME@ VERSION @VERSION@)               # VERSION = unica fonte di verità
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(FetchContent)
FetchContent_Declare(JUCE
    GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
    GIT_TAG 8.0.8 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)

juce_add_plugin(@PLUGIN_NAME@
    COMPANY_NAME             "@COMPANY@"
    BUNDLE_ID                "@BUNDLE_ID@"
    PLUGIN_MANUFACTURER_CODE @MANU_CODE@           # 4 char, congelato
    PLUGIN_CODE              @PLUGIN_CODE@          # 4 char, congelato
    FORMATS                  @FORMATS@             # es. AU VST3 Standalone
    PRODUCT_NAME             "@PRODUCT_NAME@"
    IS_SYNTH                 FALSE
    NEEDS_MIDI_INPUT         FALSE
    IS_MIDI_EFFECT           FALSE
    COPY_PLUGIN_AFTER_BUILD  TRUE
    AU_MAIN_TYPE             kAudioUnitType_Effect
    AU_SANDBOX_SAFE          TRUE
    HARDENED_RUNTIME_ENABLED FALSE)

juce_generate_juce_header(@PLUGIN_NAME@)

# Asset FINALI + layout.json (i sorgenti grezzi in assets/src/ NON vanno qui)
juce_add_binary_data(@PLUGIN_NAME@Assets SOURCES
    source/assets/layout.json
    # source/assets/background.png
    # source/assets/knob.png
    # source/assets/fader_cap.png
)

target_sources(@PLUGIN_NAME@ PRIVATE
    source/PluginProcessor.cpp
    source/PluginEditor.cpp
    source/LookAndFeel.cpp)

target_compile_definitions(@PLUGIN_NAME@ PUBLIC
    JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)

target_link_libraries(@PLUGIN_NAME@ PRIVATE
    @PLUGIN_NAME@Assets juce::juce_audio_utils juce::juce_dsp
    PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_lto_flags
           juce::juce_recommended_warning_flags)

# --- Harness screenshot headless (loop di feedback visivo) ---
juce_add_console_app(shoot)
target_sources(shoot PRIVATE ../tools/shoot/main.cpp source/PluginProcessor.cpp
                              source/PluginEditor.cpp source/LookAndFeel.cpp)
target_compile_definitions(shoot PRIVATE JUCE_STANDALONE_APPLICATION=1
    PLUGIN_PROCESSOR_CLASS=@PROCESSOR_CLASS@)
target_link_libraries(shoot PRIVATE @PLUGIN_NAME@Assets juce::juce_audio_utils
                                     juce::juce_dsp juce::juce_gui_extra)
